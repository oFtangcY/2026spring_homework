\paragraph{5-2. 推迟的场叶菲缅科(Jefimenko)公式. (1)}
\stepcounter{section}
\begin{proof}
  真空中的麦克斯韦方程
  \begin{align}
    \div{\vb{E}} &= \rho / \epsilon_0, \label{eq1-1}\\
    \curl{\vb{B}} - \frac{1}{c^2} \pdv{\vb{E}}{t} &= \mu_0 \vb{J}, \label{eq1-2}\\
    \curl{\vb{E}} + \pdv{\vb{B}}{t} &= 0, \label{eq1-3}\\
    \div{\vb{B}} &= 0. \label{eq1-4}
  \end{align}
  对 (\ref{eq1-3}) 式取旋度得到
  \begin{equation*}
    \grad(\div{\vb{E}}) - \laplacian{\vb{E}} + \pdv{t}(\curl{\vb{B}}) = 0,
  \end{equation*}
  再代入 (\ref{eq1-1}) 式和 (\ref{eq1-2}) 式，
  \begin{gather}
    \frac{1}{\epsilon_0} \grad{\rho} - \laplacian{\vb{E}} + \pdv{t} \left[\frac{1}{c^2} \pdv{\vb{E}}{t} + \mu_0 \vb{J}\right] = 0 \notag\\
    \Rightarrow \left[\laplacian - \frac{1}{c^2} \pdv[2]{t}\right] \vb{E}(\vb{x}, t) = -\frac{1}{\epsilon_0} \left[-\grad{\rho} - \frac{1}{c^2} \pdv{\vb{J}}{t}\right].
  \end{gather}
  同样地，对 (\ref{eq1-2}) 式取旋度并结合 (\ref{eq1-4}) 得到
  \begin{equation*}
    -\laplacian{\vb{B}} - \frac{1}{c^2} \pdv{t}(\curl{\vb{E}}) = \mu_0 \curl{\vb{J}},
  \end{equation*}
  代入 (\ref{eq1-3}) 式，
  \begin{equation}
    \left[\laplacian - \frac{1}{c^2} \pdv[2]{t}\right] \vb{B}(\vb{x}, t) = -\mu_0 \curl{\vb{J}}.
  \end{equation}
\end{proof}

\paragraph{(2)}
利用 Fourier 变换可以解出符合物理因果律的推迟格林函数
\begin{equation*}
  G^{(+)}(R, \tau) = \frac{\delta \left(t' - \left[t - \frac{R}{c}\right]\right)}{R},
\end{equation*}
由此可以积分得到波动方程的解为
\begin{align}
  \vb{E}(\vb{x}, t) &= \frac{1}{4\pi \epsilon_0} \int \dd[3]{\vb{x'}} \dd{t'} \frac{\delta \left(t' - \left[t - \frac{R}{c}\right]\right)}{R} \left[-\nabla' \rho - \frac{1}{c^2} \pdv{\vb{J}}{t'}\right] 
  = \frac{1}{4\pi \epsilon_0} \int \dd[3]{\vb{x'}} \frac{1}{R} \left[-\grad{\rho} - \frac{1}{c^2} \pdv{\vb{J}}{t}\right]_{\mathrm{ret}}, \\
  \vb{B}(\vb{x}, t) &= \frac{\mu_0}{4\pi} \int \dd[3]{\vb{x'}} \dd{t'} \frac{\delta \left(t' - \left[t - \frac{R}{c}\right]\right)}{R} [\nabla' \times \vb{J}]
  = \frac{\mu_0}{4\pi} \int \dd[3]{\vb{x'}} \frac{1}{R} [\nabla' \times \vb{J}]_{\mathrm{ret}}.
\end{align}

\paragraph{4. 推导长波近似下的电偶极辐射电磁场.}
书 (5.19) 式给出了长波近似下的磁矢势
\begin{equation}
  \vb{A}(\vb{x}) = -\frac{\mathrm{i} \mu_0 \omega}{4\pi} \vb{p} \frac{\mathrm{e}^{\mathrm{i} kr}}{r}. \tag{5.19}
\end{equation}
利用书 (5.14) 式以及书 (5.20) 式给出的矢量分析公式，有磁场
\begin{align}
  \vb{H} = \frac{1}{\mu_0} \curl{\vb{A}} &= -\frac{\mathrm{i} \omega}{4\pi} \curl(\vb{p} \frac{\mathrm{e}^{\mathrm{i} kr}}{r}) = -\frac{\mathrm{i} \omega}{4\pi} \grad{\frac{\mathrm{e}^{\mathrm{i} kr}}{r}} \times \vb{p} \notag\\
  &= \frac{ck^2}{4\pi} (\vb{n} \times \vb{p}) \frac{\mathrm{e}^{\mathrm{i} kr}}{r} \left(1 - \frac{1}{\mathrm{i} kr}\right).
\end{align}
进而有
\begin{align}
  \vb{E} &= \frac{\mathrm{i} Z_0}{k} \curl{\vb{H}} = \frac{\mathrm{i} k}{4\pi \epsilon_0} \curl \left[(\vb{n} \times \vb{p}) \frac{\mathrm{e}^{\mathrm{i} kr}}{r} \left(1 - \frac{1}{\mathrm{i} kr}\right)\right] \notag\\
  &= \frac{\mathrm{i} k}{4\pi \epsilon_0} \left\{\grad[\frac{\mathrm{e}^{\mathrm{i} kr}}{r} \left(1 - \frac{1}{\mathrm{i} kr}\right)] \times (\vb{n} \times \vb{p}) + \frac{\mathrm{e}^{\mathrm{i} kr}}{r} \left(1 - \frac{1}{\mathrm{i} kr}\right) \curl(\vb{n} \times \vb{p})\right\} \notag\\
  &= \frac{\mathrm{i} k}{4\pi \epsilon_0} \left\{\left(\mathrm{i}k - \frac{2}{r} + \frac{2}{\mathrm{i} kr^2}\right) \frac{\mathrm{e}^{\mathrm{i} kr}}{r} [\vb{n} \times (\vb{n} \times \vb{p})] + \frac{1}{\mathrm{i} k} \left(\frac{1}{r^3} - \frac{\mathrm{i} k}{r^2}\right) [\vb{n}(\vb{n} \cdot \vb{p}) - \vb{p}]\right\} \notag\\
  &= \frac{1}{4\pi \epsilon_0} \left\{k^2 (\vb{n} \times \vb{p}) \times \vb{n} \frac{\mathrm{e}^{\mathrm{i} kr}}{r} + [3\vb{n} (\vb{n} \cdot \vb{p}) - \vb{p}]\left(\frac{1}{r^3} - \frac{\mathrm{i} k}{r^2}\right) \mathrm{e}^{\mathrm{i} kr}\right\}.
\end{align}

\paragraph{郭书 5.10}
在直角坐标中可以将磁偶极矩表示为
\begin{equation*}
  \vb{m} = \frac{4\pi}{3} M_0 R_0^3 \mathrm{e}^{-\mathrm{i} \omega t} (\vb{e}_x + \mathrm{i} \vb{e}_y),
\end{equation*}
转换到球坐标中，
\begin{equation*}
  \vb{m} = \frac{4\pi}{3} M_0 R_0^3 \mathrm{e}^{-\mathrm{i} \omega t + \mathrm{i} \phi} (\sin \theta \vb{e}_R + \cos \theta \vb{e}_{\theta} + \mathrm{i} \vb{e}_{\phi}). 
\end{equation*}
带入辐射电磁场磁偶极项表达式
\begin{align*}
  \vb{B} &= \frac{\mu_0 \mathrm{e}^{\mathrm{i} kR}}{4\pi c^2 R} (\ddot{\vb{m}} \times \vb{e}_R) \times \vb{e}_R = \frac{\mu_0 \omega^2 R_0^3 M_0}{3c^2 R} (\vb{e}_{\theta} \cos \theta + \mathrm{i} \vb{e}_{\phi}) \mathrm{e}^{\mathrm{i}(kR - \omega t + \phi)}, \\
  \vb{E} &= -\frac{\mu_0 \mathrm{e}^{\mathrm{i} kR}}{4\pi cR} (\ddot{\vb{m}} \times \vb{e}_R) = \frac{\mu_0 \omega^2 R_0^3 M_0}{3c R} (\mathrm{i} \vb{e}_{\theta} - \vb{e}_{\phi} \cos{\theta}) \mathrm{e}^{\mathrm{i}(kR - \omega t + \phi)},
\end{align*}
故有
\begin{equation*}
  \vb{S} = \frac{1}{2\mu_0} \Re(\vb{E}^{*} \times \vb{B}) = \frac{\mu_0 \omega^4 R_0^6 M_0^2}{18 c^3 R^2} (1 + \cos^2 \theta) \vb{e}_R.
\end{equation*}

\paragraph{8-1. 运动的电偶极子.}
\paragraph{(1)}
直接计算系统的电偶极矩和磁偶极矩得到
\begin{align*}
  p'_i = \int \dd[3]{\vb{x}'} x'_i \rho'(\vb{x}') &= -\int \dd[3]{\vb{x}'} x'_i (\nabla' \cdot \vb{p}) \delta^3(\vb{x}') \\
  &= \int \dd[3]{\vb{x}'} x'_i [\delta^3(\vb{x}') \nabla' \cdot \vb{p} - \nabla' \cdot (\vb{p} \delta^3(\vb{x}'))] \\
  &= \eval{-x'_i \vb{p} \delta^3(\vb{x}')}_{\Sigma} + \int \dd[3]{\vb{x}'} (\nabla' x'_i) \cdot \vb{p} \delta^3(\vb{x}') \\
  &= \int \dd[3]{\vb{x}'} p_i \delta^3(\vb{x}') = p_i,
\end{align*}
\begin{equation*}
  \vb{m}' = \frac{1}{2} \int \dd[3]{\vb{x}'} [\vb{x}' \times \vb{J}'(\vb{x}')] = 0,
\end{equation*}
即 $K'$ 系中的电偶极矩为 $\vb{p}$，磁偶极矩为 0.

\paragraph{(2)}
K' 系中电偶极矩四矢量可以表示为 $P'^{\mu} = (P'^0, \vb{P}') = (0, \vb{p})$, K 系相对 K' 系以 $-\vb*{\beta}$ 运动，应用书 (6.8) 式可以得到
\begin{equation*}
  P^0 = \gamma (P'^0 + \vb*{\beta} \cdot \vb{P}') = \gamma \vb*{\beta} \cdot \vb{p}, \quad \vb{P} = \vb{P}' + \frac{\gamma - 1}{\vb*{\beta}^2} (\vb*{\beta} \cdot \vb{P}')\vb*{\beta} + \gamma \vb*{\beta} P'^0 = \vb{p} + \frac{\gamma - 1}{\vb*{\beta}^2} (\vb*{\beta} \cdot \vb{p}) \vb*{\beta}.
\end{equation*}
故 K 系中电偶极矩四矢量为 $P^{\mu} = (\gamma \vb*{\beta} \cdot \vb{p}, \vb{p} + \frac{\gamma - 1}{\vb*{\beta}^2} (\vb*{\beta} \cdot \vb{p}) \vb*{\beta})$.
另外， K 系中四速度为 $u_{\mu} = \gamma(1, -\vb*{\beta})$, 二者内积为
\begin{equation*}
  P \cdot u = P^{\mu} u_{\mu} = \gamma[\gamma \vb*{\beta} \cdot \vb{p} - \vb*{\beta} \cdot \vb{p} - (\gamma - 1) \vb*{\beta} \cdot \vb{p}] = 0,
\end{equation*}
即在任意 K 系中 $P^{\mu}$ 均与四速度正交.

\paragraph{(3)}
在静止系 K' 中，积分表达式可写为
\begin{equation*}
  J'^{\mu}(x) = -\int \dd{\tau} (1, 0) (\vb{p} \cdot \nabla') \delta(t' - \tau) \delta^3(\vb{x}') = (-(\vb{p} \cdot \nabla') \delta^3(\vb{x}'), 0) 
\end{equation*}
对比各项表达式即得到该协变表达式的确给出 (1) 中结果.

\paragraph{(4)}
由书 (8.8) 式，代入 $J^{\mu}(x')$ 的协变表达式得到
\begin{equation*}
  A^{\mu}(x) = \int \dd[4]{x'} D^{(+)}(x - x') J^{\mu}(x') = -4\pi \int \dd{\tau} u^{\mu}(\tau) \int \dd[4]{x'} D^{(+)}(x - x') (P \cdot \partial') \delta^4[x' - r(\tau)].
\end{equation*}
对关于 $x'$ 的项分部积分，边界项消失
\begin{equation*}
  \int \dd[4]{x'} D^{(+)}(x - x') \partial'_{\nu} \delta^4[x' - r(\tau)] = -\int \dd[4]{x'} \delta^4[x' - r(\tau)] \partial'_{\nu} D^{(+)}(x - x'),
\end{equation*}
注意到 $\partial'_{\nu} D^{(+)}(x - x') = - \partial_{\nu} D^{(+)}(x - x')$，因此我们得到
\begin{equation*}
  \int \dd[4]{x'} D^{(+)}(x - x') (P \cdot \partial') \delta^4[x' - r(\tau)] = (P \cdot \partial) \int \dd[4]{x'} \delta^4[x' - r(\tau)] D^{+}(x - x') = (P \cdot \partial) D^{(+)}[x - r(\tau)].
\end{equation*}
带回 $A^{\mu}$ 表达式
\begin{equation*}
  A^{\mu}(x) = -4\pi \int \dd{\tau} u^{\mu}(\tau) (P \cdot \partial) D^{(+)}[x - r(\tau)].
\end{equation*}
进而我们可以代入 $D^{(+)}[x - r(\tau)]$ 的具体形式
\begin{equation*}
  A^{\mu}(x) = -2 \int \dd[4]{\tau} u^{\mu}(\tau) \theta(x_0 - x'_0) (P \cdot \partial) \delta[(x - r(\tau))^2].
\end{equation*}
应用题中所给公式
\begin{equation*}
  \partial^{\nu} \delta[(x - r(\tau))^2] = \partial^{\nu} (x - r(\tau))^2 \dv{\tau}{(x - r(\tau))^2} \dv{\tau} \delta[(x - r(\tau))^2] = -\frac{(x - r(\tau))}{(x - r(\tau)) \cdot u} \dv{\tau} \delta[(x - r(\tau))^2],
\end{equation*}
带回前一个表达式得到
\begin{equation*}
  A^{\mu}(x) = 2 \int \dd[4]{\tau} u^{\mu}(\tau) \theta(x_0 - x'_0) \frac{(x - r(\tau)) \cdot P}{(x - r(\tau)) \cdot u} \dv{\tau} \delta[(x - r(\tau))^2],
\end{equation*}
再作分部积分并运用 delta 函数性质得到最终表达式
\begin{equation*}
  A^{\mu}(x) = -\int \dd{\tau} \delta[(x - r(\tau))^2] \theta(x_0 - r_0) \dv{\tau}\left[\frac{u^{\mu} \cdot (P \cdot (x - r))}{u \cdot (x - r)}\right] = -\frac{1}{u \cdot R} \dv{\tau} \left[\frac{u^{\mu (P \cdot R)}}{u \cdot R}\right]_{\mathrm{ret}},
\end{equation*}
其中 $R^{\mu} = x^{\mu} - r^{\mu}(\tau)$ 为场点到源点的相对坐标四矢量.
