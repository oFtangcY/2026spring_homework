\paragraph{1.}
\begin{flalign*}
	(1) && &\dv{x}(x^2 \dv{y}{x}) + (\lambda x + x^2) = 0, p(x) = x^2, q(x) = -x^2, w(x) = x.&&\\
	(2) && &\begin{aligned}
		&\dv{x}\left[x^a (x - 1)^{b - a} \dv{y}{x}\right] + \lambda x^{a - 1} (x - 1)^{b - a - 1} y = 0,\\
		&p(x) = x^a (x - 1)^{b - a}, q(x) = 0, w(x) = x^{a - 1} (x - 1)^{b - a - 1}.
	\end{aligned}&&\\
	(3) && &\dv{x}\left[\frac{x \mathrm{e}^{-x}}{(x - 1)^2} \dv{y}{x}\right] + \frac{\lambda \mathrm{e}^{-x}}{(x - 1)^2} y = 0, p(x) = \frac{x \mathrm{e}^{-x}}{(x - 1)^2}, q(x) = 0, w(x) = \frac{\mathrm{e}^{-x}}{(x - 1)^2}.&&\\
	(4) && &\dv{x}(\mathrm{e}^{-x^2} \dv{y}{x}) + 2\lambda \mathrm{e}^{-x^2} = 0, p(x) = \mathrm{e}^{-x^2}, q(x) = 0, w(x) = 2\mathrm{e}^{-x^2}. &&
\end{flalign*}

\paragraph{2.}
考虑 $y(x) \in D\left(\hat{L}\right), z(x) \in L^2[a, b]$，有
\begin{align*}
	\braket{z}{\hat{L} y} &= \braket{\hat{L}^{\dagger} z}{y} \\
	\Rightarrow \braket{z}{\hat{L} y} &= \int_a^b z^{*} \left\{-\dv{x}\left[p(x) \dv{y}{x}\right] + q(x) y\right\} \dd{x} \\
	&= \eval{-z^{*} p(x) \dv{y}{x}}_{a}^{b} + \int_a^b p(x) \dv{z^{*}}{x} \dv{y}{x} \dd{x} + \int_a^b q(x) z^{*}(x) y(x) \dd{x} \\
	&= \left[y(x) p(x) \dv{z^{*}(x)}{x} - z^{*}(x) p(x) \dv{y(x)}{x}\right]_{a}^{b} - \int_a^b \dv{x} \left[p(x) \dv{z^{*}}{x}\right] y(x) \dd{x} + \int_a^b q(x) z^{*}(x) y(x) \dd{x} \\
	&= \left[y(x) p(x) \dv{z^{*}(x)}{x} - z^{*}(x) p(x) \dv{y(x)}{x}\right]_{a}^{b} + \braket{\hat{L}' z}{y} = \braket{\hat{L}^{\dagger} z}{y}.
\end{align*}
若 $\hat{L}$ 的伴算符存在，即要求边界项为 0，整理并代入 $y \in D\left(\hat{L}\right)$ 满足的边界条件得到
\begin{equation*}
	y(a) \left[\frac{\cos \alpha}{c_1} z'^{*}(b) - \frac{\cos \beta}{c_2} z^{*}(b) - z'^{*}(a)\right] + y'(a)\left[-\frac{\sin \alpha}{c_1} z'^{*}(b) - \frac{\sin \beta}{c_2} z^{*}(b) + z^{*}(a)\right] = 0.
\end{equation*}
上式对任意 $y(a), y'(a)$ 成立，故伴算符存在条件为
\begin{align}
	z'(a) &= \frac{\cos \alpha}{c_1} z'(b) - \frac{\cos \beta}{c_2} z(b), \label{eq1}\\
	z(a) &= \frac{\sin \alpha}{c_1} z'(b) + \frac{\sin \beta}{c_2} z(b), \label{eq2}
\end{align}
此时 $\hat{L}$ 的伴算符为
\begin{align*}
	&\hat{L}^{\dagger} = -\dv{x}\left[p(x) \dv{x}\right] + q(x), \\
	&D\left(\hat{L}\right) = \{y(x): a \leq x \leq b, y(x) \in L^2[a, b] \text{ 且满足 (\ref{eq1}) 和 (\ref{eq2})}\},
\end{align*}
当且仅当 $\sin(\alpha + \beta) = c_1 c_2$ 时 $\hat{L}$ 是自伴算符.

\paragraph{3.}
本征函数具有形式 $\mathrm{e}^{\pm \mathrm{i} \lambda x}$，取线形组合并带入边界条件得到
\begin{align*}
	C_1 + C_2 &= 0, \\
	(1 - \beta)C_1 \mathrm{e}^{\mathrm{i} \lambda l} - (1 + \beta) C_2 \mathrm{e}^{-\mathrm{i} \lambda l} &= 0.
\end{align*}
由以上两式得到
\begin{equation*}
	2 \mathrm{i} \lambda l = \ln \frac{\beta + 1}{\beta - 1},
\end{equation*}
并考虑到对数函数的多值性与 $0 < \beta \leq 1$，取近似得到本征值
\begin{equation*}
	2 \mathrm{i} \lambda l = \ln \frac{\beta + 1}{\beta - 1} \approx 2\beta + \mathrm{i}(2n + 1)\pi \Rightarrow \lambda_n = \frac{-2 \mathrm{i} \beta + (2n + 1)\pi}{2l}, n \in \mathbb{Z}.
\end{equation*}
相应的本征函数为 $\{\mathrm{e}^{\beta x/l} \mathrm{e}^{\mathrm{i} \frac{(2n + 1)x}{2l}} - \mathrm{e}^{-\beta x/l} \mathrm{e}^{-\mathrm{i} \frac{(2n + 1)x}{2l}}\}$.

\paragraph{4.}
本征函数具有形式 $\mathrm{e}^{-\mathrm{i} \lambda x}$，带入边界条件得到
\begin{equation*}
	\mathrm{e}^{-\mathrm{i} \lambda b} = \mathrm{e}^{\mathrm{i} \theta} \mathrm{e}^{-\mathrm{i} \lambda a} \Rightarrow \lambda_n = \frac{\theta + 2n\pi}{a - b}, n \in \mathbb{Z},
\end{equation*}
相应的本征函数为 $\{\mathrm{e}^{\mathrm{i} \frac{\theta + 2n\pi}{b - a} x}\}$.
