\paragraph{1.}
\begin{proof}
	可以将 $u$ 按 $\{\Phi_k\}$ 展开，得到 $u = \sum_k c_k \Phi_k$，带回 $u$ 满足的偏微分方程即得到
	\begin{equation*}
		\sum_k \lambda_k c_k \Phi_k = f \Rightarrow \lambda_k c_k = A_k,
	\end{equation*}
	故
	\begin{equation*}
		u = \sum_k \frac{A_k}{\lambda_k} \Phi_k. \qedhere
	\end{equation*}
\end{proof}

\paragraph{2.}
对于本题中的方程及齐次条件，可以写出相应的本征值和本征函数为
\begin{align*}
	&\lambda_{mn} = \left(\frac{m\pi}{a}\right)^2 + \left(\frac{n\pi}{b}\right)^2 \\
	&w_{mn}(x, y) = \sin \frac{m\pi x}{a} \cos \frac{n\pi y}{b}
\end{align*}
将 $u(x, y)$ 按 $w_{mn}(x, y)$ 展开
\begin{equation*}
	u(x, y) = \sum_{m = 1}^{\infty} \sum_{n = 0}^{\infty} c_{mn} \sin \frac{m\pi x}{a} \cos \frac{n\pi y}{b}
\end{equation*}
并代入原定解问题的偏微分方程中得到
\begin{equation*}
	\left[\left(\frac{m\pi}{a}\right)^2 + \left(\frac{n\pi}{b}\right)^2\right] c_{mn} = \int_{0}^{b} \cos \frac{n\pi y}{b} \dd{y} \int_{0}^{a} \sin \frac{m\pi x}{a} f(x, y) \dd{x}
\end{equation*}
求解该代数方程即得到 $u(x, y)$ 的展开系数.

\paragraph{3.}
要求积分核为 $\theta$ 对应算符的本征函数，则应当取为 Legendre 函数 $P_l(\cos \theta)$.
此时 $u$ 的像函数为
\begin{equation*}
	U_l(r, \phi) = \int_{0}^{\pi} u(r, \theta, \phi) P_l(\cos\theta) \sin \theta \dd{\theta},
\end{equation*}
满足定解问题
\begin{align*}
	&\dv{r}(r^2 \dv{U_l}{r}) - l(l + 1) U_l(r) = 0 \\
	&U_l(1) = \int_{-1}^{1} f(x) P_l(x) \dd{x}
\end{align*}
一般不能对球内的 $r$ 进行积分变换，积分区间不符同时也仅有边界条件 $r = 1$ 处的 Dirichlet 边界条件，在 $r = 0$ 处仅有有界性条件.

\paragraph{4.}
可以将题述问题写为
\begin{equation*}
	\begin{cases}
		\pdv{u}{t} - \kappa \pdv[2]{u}{x} = 0, \quad -\infty < x < \infty, t > 0 \\
		\eval{u}_{x \to \pm \infty} < \infty, \quad t > 0 \\
		\eval{u}_{t = 0} = u_0 H(x), \quad -\infty < x < \infty 
	\end{cases}
\end{equation*}
作 Laplace 变换，令
\begin{equation*}
	u(x, t) \fallingdotseq U(x, p) = \int_{0}^{\infty} u(x, t) \mathrm{e}^{-pt} \dd{t},
\end{equation*}
则定解问题化为
\begin{equation*}
	pU(x, p) - \kappa \dv[2]{U(x, p)}{x} = u_0 H(x),
\end{equation*}
由此得到
\begin{equation*}
	U(x, p) = \frac{1}{2} \frac{1}{\sqrt{\kappa p}} \int_{-\infty}^{\infty} u_0 H(x') \exp\left[-\sqrt{\frac{p}{\kappa}} \abs{x - x'}\right] \dd{x'} =
	\begin{cases}
		\frac{u_0}{2p} \exp(\sqrt{\frac{p}{\kappa} x}), &x < 0, \\
		\frac{u_0}{2p} \left[2 - \exp(-\sqrt{\frac{p}{\kappa}}x)\right], &x \geq 0,
	\end{cases}
\end{equation*}
所以
\begin{equation*}
	u(x, t) = 
	\begin{cases}
		\frac{u_0}{2} \mathrm{erfc}\left(-\frac{x}{2\sqrt{\kappa t}}\right), &x < 0, \\
		u_0 \left(1 - \frac{1}{2} \mathrm{erfc} \frac{x}{2\sqrt{\kappa t}}\right), &x \geq 0.
	\end{cases}
\end{equation*}

\paragraph{5.}
对时间变量作 Laplace 变换，定解问题化为
\begin{align*}
	&p^2 U(x, p) - a^2 \dv[2]{U(x, p)}{x} = \frac{p}{p^2 + \omega^2}, \quad 0 < x < \infty \\
	&U(0, p) = 0, \quad \eval{\dv{U}{x}(x, p)}_{x \to \infty} = 0
\end{align*}
结合边界条件得到其解为
\begin{equation*}
	U(x, p) = \frac{1}{p} \frac{1}{p^2 + \omega^2} \left(1 - \mathrm{e}^{-px/a}\right),
\end{equation*}
逆变换得到原定解问题的解为
\begin{equation*}
	u(x, t) = \frac{1}{\omega^2} (1 - \cos \omega t) - \frac{1}{\omega^2} \left[1 - \cos \omega \left(t - \frac{x}{a}\right)\right] H\left(t - \frac{x}{a}\right).
\end{equation*}

\paragraph{6.}
首先求解定解问题
\begin{align*}
	&\frac{1}{\rho} \pdv{\rho}(\rho \pdv{u}{\rho}) + \pdv[2]{u}{z} = 0, \quad 0 < \rho < a, -\infty < z < \infty \\
	&\eval{u}_{\rho = a} = V_0 \mathrm{e}^{-\kappa \abs{z}} \mathrm{sgn} z, \quad -\infty < z < \infty
\end{align*}
对 $z$ 作 Fourier 变换得到关于 $U(\rho, k)$ 的定解问题
\begin{align*}
	&\frac{1}{\rho} \dv{\rho} \left[\rho \dv{U(\rho, k)}{\rho}\right] - k^2 U(\rho, k) = 0, \quad 0 < \rho < a \\
	&U(a, k) = -\frac{V_0}{\sqrt{2\pi}} \frac{2 \mathrm{i} k}{\kappa^2 + k^2}
\end{align*}
上述方程为 0 阶虚宗量 Bessel 方程，结合边界条件得到其解为
\begin{equation*}
	U(\rho, k) = -\frac{V_0}{\sqrt{2\pi}} \frac{2 \mathrm{i} k}{\kappa^2 + k^2} \frac{\mathrm{I}_0(k\rho)}{\mathrm{I}_0(ka)},
\end{equation*}
逆变换并取 $\kappa \to 0$ 得到原定解问题解为
\begin{equation*}
	u(\rho, z) = \frac{2V_0}{\pi} \int_{0}^{\infty} \frac{\mathrm{I}_0(k\rho)}{\mathrm{I}_0(ka)} \frac{\sin kz}{k} \dd{k}.
\end{equation*}
