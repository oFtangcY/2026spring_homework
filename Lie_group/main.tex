\subsubsection*{Problem 1:}
\paragraph{(1)}
There are three types of roots:
\begin{itemize}
  \item $e_i - e_j = \sum_{k = i}^{j - 1} \alpha_k$, $1 \leq i < j \leq n$;
  \item $2e_i = 2\sum_{k = i}^{n - 1} \alpha_k + \alpha_n$;
  \item $e_i + e_j = \sum_{k = i}^{j - 1} \alpha_k + 2\sum_{k = j}^{n - 1} \alpha_k + \alpha_n$, $1 \leq i < j \leq n$;
\end{itemize}
Together with their negatives form the root system of $C_n$.

\paragraph{(2)}
Three types of Weyl group elements are corresponding to the three types of roots ($x$ is an arbitrary vector $x = x_i e_i$):
\begin{itemize}
  \item $s_{e_i - e_j} x = (x_1, \cdots, x_{i - 1}, x_j, x_{i + 1}, \ldots, x_{j - 1}, x_{i}, x_{j + 1}, \ldots, x_n)$, making the reflection on $\mathbb{R}^n$ by swapping the $i$-th and $j$-th coordinates;
  \item $s_{2e_i} x = (x_1, \cdots, x_{i - 1}, -x_i, x_{i + 1}, \ldots, x_n)$, making the reflection on $\mathbb{R}^n$ by changing the sign of the $i$-th coordinate;
  \item $s_{e_i + e_j} x = (x_1, \cdots, x_{i - 1}, -x_j, x_{i + 1}, \ldots, x_{j - 1}, -x_{i}, x_{j + 1}, \ldots, x_n)$, making the reflection on $\mathbb{R}^n$ by swapping the $i$-th and $j$-th coordinates and changing the signs of both coordinates;
\end{itemize}
We can choose the first two reflections to generate the whole Weyl group.

\paragraph{(3)}
Choose the simple roots of $B_n$:
\begin{equation*}
  \beta_1 = e_1 - e_2, \beta_2 = e_2 - e_3, \ldots, \beta_{n - 1} = e_{n - 1} - e_n, \beta_n = e_n,
\end{equation*}
while the coroots are
\begin{equation*}
  H_{\beta_1} = e_1 - e_2 = \alpha_1, H_{\beta_2} = e_2 - e_3 = \alpha_2, \ldots, H_{\beta_n} = 2e_n = 2\alpha_n.
\end{equation*}
WE find the corresponding Weyl group are the same as $C_n$, where $s_{e_i} = s_{2e_i}$, so the only thing left to check is
\begin{equation*}
  \phi(s_{\alpha} \cdot e_i) = s_{\phi \alpha} \cdot (\phi e_i).
\end{equation*}
Define the linear transformation $\phi: \beta_i \rightarrow \alpha_i$, then under the Weyl group action $s_{\alpha} \cdot e_i = \pm e_i$ or $0$.
Thus
\begin{equation*}
  \phi(s_{\alpha} \cdot e_i) = \left\{
    \begin{aligned}
      \phi e_i &= 2e_i \\
      \phi (-e_i) &= -2e_i \\
      \phi 0 &= 0
    \end{aligned}
  \right\} = s_{\phi \alpha} \cdot (\phi e_i).
\end{equation*}

\paragraph{(4)}
Symmetrically, the extra white node must be $\alpha_0 = -2e_1$ and it is easy to verify the connectivity of the extended Dynkin diagram.
Thus apparently it is a non-simple root of $C_n$.

\subsubsection*{Problem 2:}
The Chevalley basis of $\mathfrak{sl}(3, \mathbb{C})$ is
\begin{gather*}
  H_{\alpha_1} = \alpha_1 = E_{11} - E_{22}, H_{\alpha_2} = \alpha_2 = E_{22} - E_{33}, \\
  E_{\alpha_1} = E_{12}, E_{\alpha_2} = E_{23}, E_{\alpha_1 + \alpha_2} = E_{13}, \\
  E_{-\alpha_1} = E_{21}, E_{-\alpha_2} = E_{32}, E_{-(\alpha_1 + \alpha_2)} = E_{31}.
\end{gather*}
It is easy to verify the Lie brackets of the Chevalley basis.

\subsubsection*{Problem 3:}
\paragraph{(1)}
First consider simple Lie algebra $\mathfrak{g}$ with $\rank{\mathfrak{g}} = 3$:
\begin{itemize}
  \item $A_3$: \dynkin{A}{3};
  \item $B_3$: \dynkin{B}{3};
  \item $C_3$: \dynkin{C}{3};
  \item $D_3$: \dynkin{D}{3} $\cong A_3$;
\end{itemize}
So non-isomorphic rank-3 simple Lie algebras are $A_3$, $B_3$ and $C_3$.
Similarly we find non-isomorphic rank-2 simple Lie algebras $A_2$, $B_2$ and $G_2$, and non-isomorphic rank-1 simple Lie algebra $A_1$.
Then we can construct semi-simple Lie algebra uding direct sum:
\begin{itemize}
  \item $A_2 \oplus A_1$;
  \item $B_2 \oplus A_1$;
  \item $G_2 \oplus A_1$;
  \item $A_1 \oplus A_1 \oplus A_1$.
\end{itemize}

\paragraph{(2)}
The dimension of the semi-simple Lie algebra is the sum of the numbers of roots and the rank, so we have
\begin{itemize}
  \item $\dim(A_3) = 15$
  \item $\dim(B_3) = 21$
  \item $\dim(C_3) = 21$
  \item $\dim(A_2 \oplus A_1) = 11$;
  \item $\dim(B_2 \oplus A_1) = 13$;
  \item $\dim(G_2 \oplus A_1) = 17$;
  \item $\dim(A_1 \oplus A_1 \oplus A_1) = 9$.
\end{itemize}

\subsubsection*{Problem 5:}
\paragraph{(1)}
The generators of $\mathfrak{so}(4)$ are $\{J^{(4)}_{ij} = E_{ij} - E_{ji} \mid 1 \leq i < j \leq 4\}$, while the generators of $\mathfrak{so}(5)$ are $\{J^{(5)}_{ij} = E_{ij} - E_{ji} \mid 1 \leq i < j \leq 5\}$.
Thus we can form $\tilde{J}^{(5)}_{ij} = \mqty{J^{(4)}_{ij} & 0 \\ 0 & 0}$, which are the generators of $\mathfrak{so}(5)$ and also forms a closed vector space.
So $\mathfrak{so}(4)$ is a subalgebra of $\mathfrak{so}(5)$.

\paragraph{(2)}
Consider the Cartan subalgebra of $\mathfrak{so}(4)$ as the $2\times 2$ block-diagonal matrices.
If we identify $\mathfrak{h}$ with $\mathbb{C}^2$ by means of the map
\begin{equation*}
  H \mapsto \mathrm{i} (a_1, a_2),
\end{equation*}
then the roots are $\pm e_1 \pm e_2$, so we can write the root system as $\{\pm \alpha_1, \pm \alpha_2\}$.
This can also be derived from $\mathfrak{so}(4) \cong D_2 \cong A_1 \oplus A_1$.

For $\mathfrak{so}(5) \cong B_2$, the root system is $\{\pm e_1, \pm e_2, \pm e_1 \pm e_2\}$.

\paragraph{(3)}
$\pm e_1 \pm e_2$ are the long roots of $B_2$, and the correspondance ensures the closure of the root system.
