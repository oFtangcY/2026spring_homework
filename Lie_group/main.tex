\paragraph{Problem 2:}
\begin{proof}
  Under the adjoint representation we can represent $\alpha_1$ as $(2, -1)$ and $\alpha_2$ as $(-1, 2)$, while with the Chevalley basis we have $\pi(X_{-\alpha_1}) = E_{-\alpha_1} = E_{21}$ and $\pi(X_{-\alpha_2}) = E_{-\alpha_2} = E_{32}$.
  Then the highest weight $\mu = (1, 1) = \alpha_1 + \alpha_2$, so the corresponding weight vector is $\comm{E_{\alpha_1}}{E_{\alpha_2}} = E_{13}$ and the multiplicity is 1.
  Acting $\pi(X_{-\alpha_1})$ and $\pi(X_{-\alpha_2})$ on $E_{13}$ we get the weight vectors $E_{23}$ of the weight $(-1, 2)$ which is $\alpha_2$, and $-E_{12}$ of the weight $(2, -1)$ which is $\alpha_1$.
  This is evident since we have the property of the Chevalley basis that $\comm{E_{\beta}}{E_{\gamma}} = N_{\beta \gamma} E_{\beta + \gamma}$.
  Thus both the two weights $\alpha_1$ and $\alpha_2$ have multiplicity 1.

  Next act $\pi(X_{-\alpha_1})$ on $-E_{12}$ and act $\pi(X_{-\alpha_2})$ on $E_{23}$, which both give the weight vector of $(0, 0)$.
  This gives two linearly independent weight vectors of $(0, 0)$: $E_{11} - E_{22} = H_{\alpha_1}$ and $E_{33} - E_{22} = -H_{\alpha_2}$.
  Hence we find the multiplicity of the weight $(0, 0)$ is 2.

  Continuously acting $\pi(X_{-\alpha_i})$ we can find the weight vectors of weights $(-2, 1) = -\alpha_1$, $(1, -2) = -\alpha_2$ and $(-1, -1) = -\alpha_1 - \alpha_2$.
  These are the opposites of the first three weights, and they also have multiplicity 1.
  Thus we have found the multiplicity of each weight which is exactly the dimension of each weight space.
\end{proof}

\paragraph{Problem 3:}
\paragraph{(1)}
For $B_2$, the positive roots are
\begin{equation*}
  \alpha_1, \alpha_2, \alpha_1 + \alpha_2, \alpha_1 + 2\alpha_2.
\end{equation*}
Thus we have
\begin{equation*}
  \delta = \frac{3}{2} \alpha_1 + 2\alpha_2.
\end{equation*}
For a general highest weight rep. with the highest weight $\mu = (m_1, m_2)$, we can compute
\begin{flalign*}
  &\innerproduct{\alpha_1}{\mu + \delta} = m_1 + 1,\quad \innerproduct{\alpha_1}{\delta} = 1, \\
  &\innerproduct{\alpha_2}{\mu + \delta} = \frac{1}{2} (m_2 + 1),\quad \innerproduct{\alpha_2}{\delta} = \frac{1}{2}, \\
  &\innerproduct{\alpha_1 + \alpha_2}{\mu + \delta} = \frac{1}{2} (2m_1 + m_2 + 1),\quad \innerproduct{\alpha_1 + \alpha_2}{\delta} = \frac{3}{2}, \\
  &\innerproduct{\alpha_1 + 2\alpha_2}{\mu + \delta} = m_1 + m_2 + 1,\quad \innerproduct{\alpha_1 + 2\alpha_2}{\delta} = 2.
\end{flalign*}
Hence,
\begin{equation*}
  \dim(\pi_{\mu}) = \frac{1}{6} (m_1 + 1)(m_2 + 1)(m_1 + m_2 + 1)(2m_1 + m_2 + 3).
\end{equation*}

\paragraph{(2)}
For $G_2$, the positive roots are
\begin{equation*}
  \alpha_1, \alpha_2, \alpha_1 + \alpha_2, \alpha_1 + 2\alpha_2, \alpha_1 + 3\alpha_2, 2\alpha_1 + 3\alpha_2.
\end{equation*}
Thus we have
\begin{equation*}
  \delta = 3\alpha_1 + 5\alpha_2.
\end{equation*}
or a general highest weight rep. with the highest weight $\mu = (m_1, m_2)$, we can compute
\begin{flalign*}
  &\innerproduct{\alpha_1}{\mu + \delta} = \frac{3}{2} (m_1 + 1),\quad \innerproduct{\alpha_1}{\delta} = \frac{3}{2}, \\
  &\innerproduct{\alpha_2}{\mu + \delta} = \frac{1}{2} (m_2 + 1),\quad \innerproduct{\alpha_2}{\delta} = \frac{1}{2}, \\
  &\innerproduct{\alpha_1 + \alpha_2}{\mu + \delta} = \frac{1}{2} (3m_1 + m_2 + 4),\quad \innerproduct{\alpha_1 + \alpha_2}{\delta} = 2, \\
  &\innerproduct{\alpha_1 + 2\alpha_2}{\mu + \delta} = \frac{1}{2} (3m_1 + 2m_2 + 5),\quad \innerproduct{\alpha_1 + 2\alpha_2}{\delta} = \frac{5}{2}, \\
  &\innerproduct{\alpha_1 + 3\alpha_2}{\mu + \delta} = \frac{1}{2} (3m_1 + 3m_2 + 6),\quad \innerproduct{\alpha_1 + 3\alpha_2}{\delta} = 3, \\
  &\innerproduct{2\alpha_1 + 3\alpha_2}{\mu + \delta} = \frac{1}{2} (6m_1 + 3m_2 + 9),\quad \innerproduct{2\alpha_1 + 3\alpha_2}{\delta} = \frac{9}{2}.
\end{flalign*}
Hence,
\begin{equation*}
  \dim(\pi_{\mu}) = \frac{1}{120} (m_1 + 1)(m_2 + 1)(m_1 + m_2 + 2)(2m_1 + m_2 + 3)(3m_1 + m_2 + 4)(3m_1 + 2m_2 + 5).
\end{equation*}

\paragraph{(2)}
The direct product can be rewritten as
\begin{equation*}
  (\boldsymbol{3}, \boldsymbol{2}) \otimes (\bar{\boldsymbol{3}}, \boldsymbol{1}) \otimes (\bar{\boldsymbol{1}}, \boldsymbol{2}) = (\boldsymbol{3} \otimes \bar{\boldsymbol{3}} \otimes \boldsymbol{1}, \boldsymbol{2} \otimes \boldsymbol{1} \otimes \boldsymbol{2}).
\end{equation*}
We have the decompositions $\boldsymbol{3} \otimes \bar{\boldsymbol{3}} = \boldsymbol{8} \oplus \boldsymbol{1}$ and $\boldsymbol{2} \otimes \boldsymbol{2} = \boldsymbol{3} \oplus \boldsymbol{1}$, so we can further decompose the above tensor product as
\begin{equation*}
  (\boldsymbol{3}, \boldsymbol{2}) \otimes (\bar{\boldsymbol{3}}, \boldsymbol{1}) \otimes (\bar{\boldsymbol{1}}, \boldsymbol{2}) = (\boldsymbol{8} \oplus \boldsymbol{1}, \boldsymbol{3} \oplus \boldsymbol{1}) = (\boldsymbol{8}, \boldsymbol{3}) \oplus (\boldsymbol{8}, \boldsymbol{1}) \oplus (\boldsymbol{1}, \boldsymbol{3}) \oplus (\boldsymbol{1}, \boldsymbol{1}).
\end{equation*}

\paragraph{Problem 5:}
\paragraph{(1)}
First we can define a tensor by contracting one of the symmetric indices with the third index:
\begin{equation*}
  M^{ijk} = \epsilon^{ik} \epsilon_{lm} T^{(jl)m} + \epsilon^{jk} \epsilon_{lm} T^{(il)m}
\end{equation*}
which remains symmetric in $i$ and $j$.
We can compute each component of $M^{ijk}$
\begin{equation*}
  M^{111} = 0, \quad M^{222} = 0, \quad M^{112} = -M^{211} = T^{(11)2} - T^{(12)1}, \quad M^{122} = -M^{221} = T^{(12)2} - T^{(22)1},
\end{equation*}
so it describes a 2-diemnsional invariant subspace.

Thus we can divide $T^{(ij)k}$ into two parts:
\begin{equation*}
  T^{(ij)k} = (T^{(ij)k} - xM^{ijk}) + xM^{ijk},
\end{equation*}
the first part with components
\begin{align*}
  &S^{111} = T^{(11)1},\quad S^{222} = T^{(22)2}, \\
  &S^{112} = (1 - x) T^{(11)2} + x T^{(12)1}, \quad S^{211} = (1 - x) T^{(12)1} + xT^{(11)2}, \\
  &S^{122} = (1 - x) T^{(12)2} + x T^{(22)1}, \quad S^{211} = (1 - x) T^{(22)1} + xT^{(12)2}.
\end{align*}
Choosing $x = \frac{1}{3}$, we can write the components as
\begin{equation*}
  S^{(ijk)} = \frac{1}{3} (T^{(ij)k} + T^{(jk)i} + T^{(ki)j}),
\end{equation*}
which is totally symmetric in $i, j, k$, forming a 4-dimensional invariant subspace.
Hence we can write the decomposition
\begin{equation*}
  \boldsymbol{3} \otimes \boldsymbol{2} = \boldsymbol{4} \oplus \boldsymbol{2}.
\end{equation*}

\paragraph{Problem 6:}
\paragraph{(1)}
The weight diagram of $\boldsymbol{15}$ of $\mathfrak{su}(4)$ is showed in Problem 1(3), and only considering the first two coordinates we can divide the daigram into four parts:
upper-left three weights is the rep. $\boldsymbol{3}$ of $\mathfrak{su}(3)$, lower-right three is the rep. $\bar{\boldsymbol{3}}$ of $\mathfrak{su}(3)$, the other forms the rep. $\boldsymbol{8}$ and the rest $(0, 0)$ is the trivial rep. of $\mathfrak{su}(3)$.
Hence we can write the decomposition
\begin{equation*}
  \boldsymbol{15} = \boldsymbol{8} \oplus \boldsymbol{3} \oplus \bar{\boldsymbol{3}} \oplus \boldsymbol{1}.
\end{equation*}
The generator of $\mathfrak{u}(1)$ can be chosen as $Q = H_{\alpha_1} + 2H_{\alpha_2} + 3H_{\alpha_3}$, so the final branching rule is then
\begin{equation*}
  \boldsymbol{15} \rightarrow \boldsymbol{8}_{0} \oplus \boldsymbol{3}_{4} \oplus \bar{\boldsymbol{3}}_{-4} \oplus \boldsymbol{1}_{0}.
\end{equation*}

\paragraph{(2)}
The weight diagram of $\boldsymbol{6}$ of $\mathfrak{su}(4)$ is showed in Problem 1(2).
Ignoring the middle coordinate we can find the upper one and the lower one are both trivial reps. of $\mathfrak{su}(2)$, while the middle four form the rep. $(\boldsymbol{2}, \boldsymbol{2})$ of $\mathfrak{su}(2)$.
Hence the decomposition
\begin{equation*}
  \boldsymbol{6} = (\boldsymbol{2}, \boldsymbol{2}) \oplus (\boldsymbol{1}, \boldsymbol{1}) \oplus (\boldsymbol{1}, \boldsymbol{1}).
\end{equation*}
The generator of $\mathfrak{u}(1)$ can be chosen as $Q = H_{\alpha_1} + 2H_{\alpha_2} + H_{\alpha_3}$, so the final branching rule is
\begin{equation*}
  \boldsymbol{6} \rightarrow (\boldsymbol{2}, \boldsymbol{2})_{0} \oplus (\boldsymbol{1}, \boldsymbol{1})_{2} \oplus (\boldsymbol{1}, \boldsymbol{1})_{-2}.
\end{equation*}

\paragraph{(3)}
For the fundamental rep. $\boldsymbol{5}$ of $\mathfrak{su}(5)$, its weights are
\begin{equation*}
  (1, 0, 0, 0) \quad (-1, 1, 0, 0),\quad (0, -1, 1, 0),\quad (0, 0, -1, 1),\quad (0, 0, 0, -1).
\end{equation*}
only consider the second and the fourth cooredinates we can write the decomposition
\begin{equation*}
  \boldsymbol{5} = (\boldsymbol{1}, \boldsymbol{1}) \oplus (\boldsymbol{2}, \boldsymbol{1}) \oplus (\boldsymbol{1}, \boldsymbol{2}).
\end{equation*}
The generators can be chosen as $Q_1 = H_{\alpha_2} + 2H_{\alpha_3} + H_{\alpha_4}$ and $Q_2 = 4H_{\alpha_1} + 3H_{\alpha_2} + 2H_{\alpha_3} + H_{\alpha_4}$, and the final branching rule is then
\begin{equation*}
  \boldsymbol{5} \rightarrow (\boldsymbol{1}, \boldsymbol{1})_{(0, 4)} \oplus (\boldsymbol{2}, \boldsymbol{1})_{1, -1} \oplus (\boldsymbol{1}, \boldsymbol{2})_{(-1, -1)}.
\end{equation*}
